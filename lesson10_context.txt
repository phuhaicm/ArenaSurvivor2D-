using System;
using UnityEngine;

public class PracticeBestScore : HaiMonoBehaviour
{
    private const string BestScoreKey =
        "PracticeBestScore";

    [Header("References")]
    [SerializeField]
    private PracticeRunScore runScore;

    private int bestScore;

    public event Action<int> BestScoreChanged;

    public int BestScore => bestScore;

    protected override void LoadComponents()
    {
        base.LoadComponents();

        LoadRunScore();
    }

    private void LoadRunScore()
    {
        if (runScore != null)
        {
            return;
        }

        runScore =
            FindFirstObjectByType<PracticeRunScore>();
    }

    protected override void OnEnable()
    {
        base.OnEnable();

        LoadRunScore();
        SubscribeScoreEvents();
    }

    protected override void OnDisable()
    {
        UnsubscribeScoreEvents();

        base.OnDisable();
    }

    protected override void Start()
    {
        base.Start();

        LoadBestScore();
    }

    // =========================================================
    // RUN SCORE EVENTS
    // =========================================================

    private void SubscribeScoreEvents()
    {
        if (runScore == null)
        {
            Debug.LogError(
                $"{name}: Run Score reference is missing.",
                this
            );

            return;
        }

        runScore.ScoreCalculated -=
            HandleScoreCalculated;

        runScore.ScoreCalculated +=
            HandleScoreCalculated;
    }

    private void UnsubscribeScoreEvents()
    {
        if (runScore == null)
        {
            return;
        }

        runScore.ScoreCalculated -=
            HandleScoreCalculated;
    }

    private void HandleScoreCalculated(int finalScore)
    {
        if (finalScore <= bestScore)
        {
            return;
        }

        // Cập nhật dữ liệu nội bộ trước.
        bestScore = finalScore;

        // Sau đó lưu dữ liệu.
        SaveBestScore();

        // Cuối cùng mới thông báo cho subscriber.
        BestScoreChanged?.Invoke(bestScore);
    }

    // =========================================================
    // SAVE / LOAD
    // =========================================================

    private void LoadBestScore()
    {
        bestScore =
            PlayerPrefs.GetInt(
                BestScoreKey,
                0
            );//playerprefs được dùng để lưu các dữ liệu nhỏ int float string phù hợp với best score, âm lượng, một vài tùy chọn cài đặt đơn giản

        Debug.Log(
            $"{name}: Best Score loaded: {bestScore}",
            this
        );

        BestScoreChanged?.Invoke(bestScore);
    }

    private void SaveBestScore()
    {
        PlayerPrefs.SetInt(
            BestScoreKey,
            bestScore
        );

        PlayerPrefs.Save();

        Debug.Log(
            $"{name}: New Best Score saved: {bestScore}",
            this
        );
    }
}

using System;
using UnityEngine;

public class PracticeRunScore : HaiMonoBehaviour
{
    [Header("Reference")]
    [SerializeField] protected PracticePlayerHealth playerHealth;
    [SerializeField] protected PracticeSurvivalTimer survivalTimer;
    [SerializeField] protected PracticePlayerKillCounter playerKillCounter;

    [Header("Score Settings")]
    [SerializeField, Min(0)] protected int pointPerSeconds = 10;
    [SerializeField, Min(0)] protected int pointPerKills = 100;
    public event Action<int> ScoreCalculated;
    protected int finalScore;
    protected bool isScore;
    public int FinalScore => finalScore;
    public bool IsScore => isScore;
    protected override void LoadComponents()
    {
        base.LoadComponents();
        
        LoadPlayerHealth();
        LoadSurvivalTimer();
        LoadPlayerKillCounter();
    }
    protected void LoadPlayerHealth()
    {
        if(playerHealth != null) return;

        playerHealth = FindFirstObjectByType<PracticePlayerHealth>();
    }
    protected void LoadSurvivalTimer()
    {
        if(survivalTimer != null) return;

        survivalTimer = FindFirstObjectByType<PracticeSurvivalTimer>();
    }
    protected void LoadPlayerKillCounter()
    {
        if(playerKillCounter != null) return;

        playerKillCounter = FindFirstObjectByType<PracticePlayerKillCounter>();
    }
    protected override void Start()
    {
        base.Start();
        

        finalScore = 0;
        isScore = false;
        ScoreCalculated?.Invoke(finalScore);
    }
    protected override void OnEnable()
{
    base.OnEnable();

    Debug.Log(
        $"{name}: OnEnable | " +
        $"PlayerHealth = {playerHealth}",
        this
    );

    LoadPlayerHealth();

    Debug.Log(
        $"{name}: Sau LoadPlayerHealth | " +
        $"PlayerHealth = {playerHealth}",
        this
    );

    SubscribePlayerEvents();
}

    protected override void OnDisable()
    {
        UnSubscribePlayerEvents();
        
        base.OnDisable();
    }
    protected void SubscribePlayerEvents()
    {
        if(playerHealth == null) return;
        playerHealth.Died += HandlePlayerDied;
    }

    protected void UnSubscribePlayerEvents()
    {
        if(playerHealth == null) return;
        playerHealth.Died -= HandlePlayerDied;
    }
    protected void HandlePlayerDied()
    {
        CalcutedScore();
    }
    protected void CalcutedScore()
    {
        if(isScore) return;
        if(survivalTimer == null) return;
        if(playerKillCounter == null) return;

        int survivalSecond = Mathf.FloorToInt(survivalTimer.ElapsedTime);
        int currentKills = playerKillCounter.CurrentKills;
        int timeScore = survivalSecond * pointPerSeconds;
        int killScore = currentKills * pointPerKills;
        finalScore = timeScore * killScore;

        isScore = true;
        ScoreCalculated?.Invoke(finalScore);
    }



}

using UnityEngine;
using System;
public class PracticeSurvivalTimer : HaiMonoBehaviour
{
    [Header("References")]
    [SerializeField] private PracticePlayerHealth playerHealth;
    [SerializeField] protected float elapsedTime;
    [SerializeField] protected bool isRunning;
    public event Action<float> TimeChanged; 
    public event Action<float> SurvivalTime; 
    public float ElapsedTime => elapsedTime;
    public bool Isrunning => isRunning;
    protected override void LoadComponents()
    {
        base.LoadComponents();
        LoadPlayerHealth();
    }
    protected void LoadPlayerHealth()
    {
        if(playerHealth != null)//khác null là đã gắn rồi còn null là chưa gắn mới đi tìm gắn vào 
        {
            return;
        }
        playerHealth = FindFirstObjectByType<PracticePlayerHealth>();
    }
    protected override void OnEnable()
    {
        base.OnEnable();
        SubscribePlayerEvents();
    }
    protected override void OnDisable()
    {
        UnSubscribePlayerEvents();
        base.OnDisable();
    }
    protected override void Start()
    {
        base.Start();
        StartTimer();
    }
    protected void Update()
    {
        if(!isRunning) return;

        elapsedTime += Time.deltaTime;//chạy theo frame với lại khi dừng để upgrade thì time sẽ ko chạy
        TimeChanged?.Invoke(elapsedTime);
    }
    protected void SubscribePlayerEvents()
    {
        if(playerHealth == null) return;
        playerHealth.Died += HandlePlayerDied;
    }
    protected void UnSubscribePlayerEvents()
    {
        if(playerHealth == null) return;
        playerHealth.Died -= HandlePlayerDied;
    }
    protected void StartTimer()
    {
        elapsedTime = 0f;
        isRunning = true;

        TimeChanged?.Invoke(elapsedTime);//phải phát thì UI mới reset timer về 0 được còn ko thì sẽ sài trong inspector
    }
    protected void HandlePlayerDied()
    {
        StopTimer();
    }
    protected void StopTimer()
    {
        if(!isRunning) return;
        isRunning = false;

        TimeChanged?.Invoke(elapsedTime);//đảm bảo UI nhận đúng giá trị cuối cùng
        SurvivalTime?.Invoke(elapsedTime);//thông báo lượt chơi đã kết thúc
    }
}

        LoadSceneLoader();

    {
        base.OnEnable();

        LoadPlayerHealth();
        LoadGameStateController();

        SubscribePlayerEvents();
        SubscribeButtonEvents();
        SubscribeGameStateEvents();
    }

    protected override void OnDisable()
    {
        UnsubscribePlayerEvents();
        UnsubscribeButtonEvents();
        UnSubscribeGameStateEvents();

        base.OnDisable();
    }

    protected override void Start()
    {
        base.Start();

        isGameOverHandled = false;
        HideGameOverPanel();
    }

    // =========================================================
    // PLAYER EVENT
    // =========================================================

    private void SubscribePlayerEvents()
    {
        if (playerHealth == null)
        {
            Debug.LogError(
                $"{name}: Player Health đang null.",
                this
            );

            return;
        }

        playerHealth.Died -= HandlePlayerDied;
        playerHealth.Died += HandlePlayerDied;
    }

    private void UnsubscribePlayerEvents()
    {
        if (playerHealth == null)
        {
            return;
        }

        playerHealth.Died -= HandlePlayerDied;
    }

    private void HandlePlayerDied()
    {
        if (isGameOverHandled)
        {
            return;
        }

        if (gameStateController == null)
        {
            Debug.LogError(
                $"{name}: Game State Controller đang null.",
                this
            );

            return;
        }

        bool gameOverStarted =
            gameStateController.RequestGameOver();

        if (!gameOverStarted &&
            !gameStateController.IsGameOver)
        {
            return;
        }

        isGameOverHandled = true;

        //ShowGameOverPanel();

        Debug.Log(
            $"{name}: Game Over started.",
            this
        );
    }

    // =========================================================
    // BUTTON EVENTS
    // =========================================================

    private void SubscribeButtonEvents()
    {
        if (restartButton != null)
        {
            restartButton.onClick.RemoveListener(
                HandleRestartButtonClicked
            );

                HandleRestartButtonClicked
            );
