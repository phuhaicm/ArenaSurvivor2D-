using UnityEngine;

public static class PracticeNearestEnemyFinder
{
    public static PracticeEnemyHealth FindNearest(Vector2 origin) 
    {
        PracticeEnemyHealth[] enemies = Object.FindObjectsOfType<PracticeEnemyHealth>();

        PracticeEnemyHealth nearest = null;
        float nearestDistance = float.MaxValue;
        foreach (PracticeEnemyHealth enemy in enemies)
        {
            if (enemy == null) continue;
            // tiếp tục là bởi khi nếu enemy này null thì tiếp tục enemy tiếp theo
            float distance = Vector2.Distance(origin, enemy.transform.position);

            if (distance < nearestDistance)
            {
                nearestDistance = distance;
                nearest = enemy;
            }
        }
        return nearest;

    }
}

using UnityEngine;

public class PracticeLevelUpListener : HaiMonoBehaviour
{
    [SerializeField] protected PracticePlayerLevelSystem levelSystem;

    protected override void LoadComponents()
    {
        base.LoadComponents();
        LoadLevelSystem();
    }

    protected virtual void LoadLevelSystem()
    {
        if (levelSystem != null) return;

        levelSystem = GetComponent<PracticePlayerLevelSystem>();
    }

    protected override void OnEnable()
    {
        base.OnEnable();
        SubscribeEvents();
    }

    protected override void OnDisable()
    {
        base.OnDisable();
        UnsubscribeEvents();
    }

    protected virtual void SubscribeEvents()
    {
        if (levelSystem == null) return;

        levelSystem.LevelChanged += HandleLeveledUp;
    }

    protected virtual void UnsubscribeEvents()
    {
        if (levelSystem == null) return;

        levelSystem.LevelChanged -= HandleLeveledUp;
    }

    protected virtual void HandleLeveledUp(float newLevel)
    {
        Debug.Log(
            $"Received Level Up Event | New Level: {newLevel}",
            gameObject
        );
    }
}

using TMPro;
using UnityEngine;
using System.Collections;

public class PracticeShooterStatsDebugUI : HaiMonoBehaviour
{
    [SerializeField] protected PracticeShooterWithDamage shooter;
    [SerializeField] protected TextMeshProUGUI statsText;

    protected override void LoadComponents()
    {
        base.LoadComponents();

        if (shooter == null)
        {
            shooter = GetComponent<PracticeShooterWithDamage>();
        }

        if (statsText == null)
        {
            statsText = GetComponentInChildren<TextMeshProUGUI>();
        }
    }

    private void Update()
    {
        if (shooter == null || statsText == null) return;

        statsText.text =
            $"Damage: {shooter.BulletDamage}\n" +
            $"Fire Rate: {shooter.ShootInterval}\n" +
            $"Projectiles: {shooter.ProjectileCount}";
    }
}

using UnityEngine;
using System;

public class PracticePlayerKillCounter : HaiMonoBehaviour
{
    [Header("References")]
    [SerializeField] protected PracticeEnemySpawner enemySpawner;
    [SerializeField] protected PracticePlayerHealth playerHealth;
    [SerializeField] private int currentKills;
    [SerializeField] private bool isCounting;
    public event Action<int> KillCountChanged;
    public event Action<int> KillCountEnded;
    public int CurrentKills => currentKills;
    public bool IsCounting => isCounting;

    protected override void LoadComponents()
    {
        base.LoadComponents();
        LoadEnemySpawner();
        LoadPlayerHealth();
    }
    protected void LoadEnemySpawner()
    {
        if(enemySpawner != null)//khác null là đã gắn rồi còn null là chưa gắn mới đi tìm gắn vào 
        {
            return;
        }
        enemySpawner = FindFirstObjectByType<PracticeEnemySpawner>();
    }
    protected void LoadPlayerHealth()
    {
        if(playerHealth != null)//khác null là đã gắn rồi còn null là chưa gắn mới đi tìm gắn vào 
        {
            return;
        }
        playerHealth = FindFirstObjectByType<PracticePlayerHealth>();
    }
    protected override void OnEnable()
    {
        base.OnEnable();
        SubscribeSpawnerEvents();
        SubscribePlayerEvents();
    }
    protected override void OnDisable()
    {
        UnSubscribeSpawnerEvents();
        UnSubscribePlayerEvents();
        base.OnDisable();
    }
    protected override void Start()
    {
        base.Start();
        StartCounting();
    }
    protected void StartCounting()
    {
        currentKills = 0;
        isCounting = true;
        
        KillCountChanged?.Invoke(currentKills);
    }
    protected void SubscribePlayerEvents()
    {
        if(playerHealth == null) return;
        playerHealth.Died += HandlePlayerDied;
    }
    protected void UnSubscribePlayerEvents()
    {
        if(playerHealth == null) return;
        playerHealth.Died -= HandlePlayerDied;
    }
    protected void SubscribeSpawnerEvents()
    {
        if(enemySpawner == null) return;
        enemySpawner.EnemyDefeated += HandleEnemyDefeated;
    }
    protected void UnSubscribeSpawnerEvents()
    {
        if(enemySpawner == null) return;
        enemySpawner.EnemyDefeated -= HandleEnemyDefeated;
    }
    protected void HandleEnemyDefeated()
    {
        if(!isCounting) return;
        
        currentKills++;
        KillCountChanged?.Invoke(currentKills);
    }
    protected void HandlePlayerDied()
    {
        StopCounting();
    }
    protected void StopCounting()
    {
        if(!isCounting) return;

        isCounting = false;
        KillCountChanged?.Invoke(currentKills);
        KillCountEnded?.Invoke(currentKills);
    }
}
