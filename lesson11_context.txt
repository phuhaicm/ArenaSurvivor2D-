using System;
using UnityEngine;

public class PracticeDifficultySystem : HaiMonoBehaviour
{
    [Header("Difficulty Timing")] 
    [SerializeField, Min(1f)] private float difficultyInterval = 30f;
    [Header("Spawn Interval Difficulty")] 
    [SerializeField, Min(0.1f)] private float initialSpawnInterval = 2f;
    [SerializeField, Min(0f)] private float spawnIntervalReduction = 0.2f;// mỗi level giảm 0.2
    [SerializeField, Min(0.05f)] private float minimunSpawnInterval = 0.5f;//giới hạn nhỏ nhất vì cứ giảm mãi thì sẽ bị âm
    [Header("Max Alive Difficulty")] 
    [SerializeField, Min(1f)] private int initialMaxAliveEnemies = 5;
    [SerializeField, Min(0)] private int maxAliveIncreasePerLevel = 2;
    [SerializeField, Min(1)] private int maximunAliveEnemies = 30;
    [Header("Batch Difficulty")] 
    [SerializeField, Min(1f)] private int initialBatchSize = 1;
    [SerializeField, Min(0)] private int maxBatchIncreasePerLevel = 1;
    [SerializeField, Min(1)] private int maximunBatchSize = 5;
    private int currentDifficultyLevel = 1;
    private float nextDifficultyTime;
    public event Action<PracticeDifficultyData> DifficultyChanged;
    public int CurrentDifficultyLevel => currentDifficultyLevel;

    protected override void Start()
    {
        base.Start();
        currentDifficultyLevel = 1;
        nextDifficultyTime = Time.time + difficultyInterval;

        PublishCurrentlyDifficulty();
    }
    private void Update() {
        UpdateDifficultyOverTime();
    }

    private void UpdateDifficultyOverTime()
    {
        bool difficultyInCreased = false;
        while(Time.time >= nextDifficultyTime)
        {
            currentDifficultyLevel++;
            nextDifficultyTime += difficultyInterval;
            difficultyInCreased = true;
        }
        if (difficultyInCreased)
        {
        PublishCurrentlyDifficulty();
        }
    }
    private PracticeDifficultyData CreateCurrentDifficultyData()
    {
        int difficultyIncreaseCount = currentDifficultyLevel - 1;
        float calculatedSpawnInterval = initialSpawnInterval - spawnIntervalReduction * difficultyIncreaseCount;
        float currentSpawnInterval = Mathf.Max(minimunSpawnInterval, calculatedSpawnInterval);

        int calculatedMaxAliveEnemies = initialMaxAliveEnemies + maxAliveIncreasePerLevel * difficultyIncreaseCount;
        int currentMaxAliveEnemis = Mathf.Min(maximunAliveEnemies, calculatedMaxAliveEnemies);

        int calculatedBatchSize = initialBatchSize + maxBatchIncreasePerLevel * difficultyIncreaseCount;
        int currentBatchSize = Mathf.Min(maximunBatchSize, calculatedBatchSize);

        return new PracticeDifficultyData(currentDifficultyLevel, currentSpawnInterval, currentMaxAliveEnemis, currentBatchSize);
    }
    private void PublishCurrentlyDifficulty()
    {
        PracticeDifficultyData difficultyData = CreateCurrentDifficultyData();

        DifficultyChanged?.Invoke(difficultyData);
        Debug.Log(
            $"Difficulty Level: " + $"{difficultyData.Level} | " + $"Spawn Interval: " + $"{difficultyData.SpawnInterval:F2} | "
            + $"Max Alive: " + $"{difficultyData.MaxAliveEnemies}" + $"Batch Size: " + $"{difficultyData.BatchSize}"
        );
    }
    private void OnValidate()
    {
        if(minimunSpawnInterval < initialSpawnInterval)
        {
            Debug.Log(
                $"{name}: Minium Spawn Interval " + "khong nen lon hon Spawn Interval. ", this
            );
        }
        if(maximunAliveEnemies < initialMaxAliveEnemies)
        {
            Debug.Log(
                $"{name}: Maxium Alive Enemies " + "khong nen nho hon initial alive enemies. ", this
            );
        }
    }
}

using System.Collections;
using TMPro;
using UnityEngine.UI;
using UnityEngine.SceneManagement;
using UnityEngine;

public class PracticeSceneLoader : HaiMonoBehaviour
{
    [Header("Loading UI")]
    [SerializeField] protected GameObject loadingPanel;
    [SerializeField] protected CanvasGroup loadingCanvasGroup;
    [SerializeField] protected Slider progressSlider;
    [SerializeField] protected TextMeshProUGUI progressText;
    [SerializeField] protected static PracticeSceneLoader instance;
    [Header("Loading Setting")]
    [SerializeField, Min(0f)] protected float miniumLoadingDuration = 1f;
    [Header("Fade Setting")]
    [SerializeField, Min(0f)] protected float fadeDuration = 0.35f;
    public static PracticeSceneLoader Instance => instance;
    protected bool isLoading;
    protected bool IsLoading => isLoading;
    protected override void Awake()
    {
        base.Awake();

        if(instance != null && instance != this)
        {
            Destroy(gameObject);
            return;
        }
        instance = this;// vì khi instance null thì nó chọn hiện tại nên là this và kể từ đó destroy những object sau

        DontDestroyOnLoad(gameObject);
    }
    protected override void Start()
    {
        base.Start();

        isLoading = false;
        
        InitializeLoadingUI();
    }
    protected void InitializeLoadingUI()
    {
        RefreshLoadingUI(0f);

        if(loadingCanvasGroup != null)
        {
            loadingCanvasGroup.alpha = 0f;
            loadingCanvasGroup.interactable = false;
            loadingCanvasGroup.blocksRaycasts = false;
        }
        SetLoadingPanelActive(false);
    }
    public bool LoadScene(string sceneName)
    {
        if(isLoading) return false;

        if (string.IsNullOrWhiteSpace(sceneName))
        {
            Debug.LogError($"{name}: Scene name is empty.", this);
            return false;
        }
        StartCoroutine(LoadSceneRoutine(sceneName));
        return true;
    }
    protected IEnumerator FadeLoadingPanelIn()
    {
        if(loadingCanvasGroup == null)
        {
            yield break;
        }
        loadingCanvasGroup.alpha = 0f;
        loadingCanvasGroup.interactable = true;
        loadingCanvasGroup.blocksRaycasts = true;

        if(fadeDuration <= 0f)
        {
            loadingCanvasGroup.alpha = 1f;
            yield break;
        }
        float elapsedTime = 0f;

        while(elapsedTime < fadeDuration)
        {
            elapsedTime += Time.unscaledDeltaTime;
            float fadeProgress = Mathf.Clamp01(elapsedTime / fadeDuration);

            loadingCanvasGroup.alpha = fadeProgress;
            yield return null;
        }
        loadingCanvasGroup.alpha = 1f;
    }
    protected IEnumerator FadeLoadingPanelOut()
    {
        if(loadingCanvasGroup == null) yield break;

        loadingCanvasGroup.alpha = 1f;

        if(fadeDuration <= 0f)
        {
            loadingCanvasGroup.alpha = 1f;
            yield break;
        }
        float elapsedTime = 0f;
        while(elapsedTime < fadeDuration)
        {
            elapsedTime += Time.unscaledDeltaTime;
            float fadeProgress = Mathf.Clamp01(elapsedTime / fadeDuration);

            loadingCanvasGroup.alpha = 1f - fadeProgress;// này là fade out nên cần ngược lại
            yield return null;
        }
        loadingCanvasGroup.alpha = 0f;
    }
    protected IEnumerator LoadSceneRoutine(string sceneName)
    {
        isLoading = true;

        Time.timeScale = 1f;
        SetLoadingPanelActive(true);
        RefreshLoadingUI(0f);
        yield return FadeLoadingPanelIn();//chờ fade in hoàn thành sau đó mới tiếp tục load scene

        float loadingStartTime = Time.unscaledTime;

        AsyncOperation loadOperation = SceneManager.LoadSceneAsync(sceneName);

        if(loadOperation == null)
        {
            Debug.LogError($"{name}: Cannot load scene {sceneName}", this);
            
            isLoading = false;
            SetLoadingPanelActive(false);
            yield break;
        }

        loadOperation.allowSceneActivation = false;

        while (!loadOperation.isDone)
        {
            float sceneProgress = Mathf.Clamp01(loadOperation.progress / 0.9f);//kẹp hoặc giới hạn 1 giá trị vào 1 khoảng nhất định

            float elapsedLoadingTime = Time.unscaledTime - loadingStartTime;//time.unscaledTime là số giây đã trôi qua kể từ khi game bắt đầu những ko ảnh hưởng bởi time.timescale 
            float timeProgress = 1f;

            if(miniumLoadingDuration > 0f)
            {
                timeProgress = Mathf.Clamp01(elapsedLoadingTime / miniumLoadingDuration);
            }

            float displayProgress = Mathf.Min(sceneProgress, timeProgress);//ta ko muốn UI chạy nhanh hơn điều kiện còn chậm
            

            RefreshLoadingUI(displayProgress);

            bool sceneIsReady = loadOperation.progress >= 0.9f;
            bool miniumTimeComplete = elapsedLoadingTime >= miniumLoadingDuration;

            if(sceneIsReady && miniumTimeComplete)
            {
                RefreshLoadingUI(1f);
                loadOperation.allowSceneActivation = true;
            }
            yield return null;//cái này mà nằm ngoài thì sẽ bị vòng lặp vô hạn
        }
        yield return FadeLoadingPanelOut();
        CompleteLoading();
    }
    protected void CompleteLoading()
    {
        isLoading = false;

        if(loadingCanvasGroup != null)
        {
            loadingCanvasGroup.alpha = 0f;
            loadingCanvasGroup.interactable = false;
            loadingCanvasGroup.blocksRaycasts = false;
        }
        SetLoadingPanelActive(false);
    }
    protected void RefreshLoadingUI(float progress)
    {
        //if(progress == null) return;
        float validProgress = Mathf.Clamp01(progress);

        if(progressSlider != null)
        {
            progressSlider.value = validProgress;
        }
        if(progressText != null)
        {
            int percentage = Mathf.RoundToInt(validProgress * 100f);

            progressText.text = $"Loading ... {percentage}%";
        }
    }
    protected void SetLoadingPanelActive(bool isActive)
    {
        if(loadingPanel == null) return;

        loadingPanel.SetActive(isActive);
    }
}
