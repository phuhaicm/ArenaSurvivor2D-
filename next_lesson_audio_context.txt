using UnityEngine;

public class PracticeSceneAudioTrigger : HaiMonoBehaviour
{
    [Header("Gameplay System References")]
    [SerializeField]
    private PracticeGameStateController gameStateController;

    [SerializeField]
    private PracticeRunDirector runDirector;

    [SerializeField]
    private PracticeFinalBossController finalBossController;

    private bool hasTriggeredBossMusic;

    protected override void LoadComponents()
    {
        base.LoadComponents();
        LoadGameplaySystems();
    }

    protected override void OnEnable()
    {
        base.OnEnable();

        hasTriggeredBossMusic = false;
        LoadGameplaySystems();
        SubscribeEvents();
    }

    protected override void OnDisable()
    {
        UnsubscribeEvents();
        base.OnDisable();
    }

    private void LoadGameplaySystems()
    {
        if (gameStateController == null)
        {
            gameStateController = FindFirstObjectByType<PracticeGameStateController>();
        }

        if (runDirector == null)
        {
            runDirector = FindFirstObjectByType<PracticeRunDirector>();
        }

        if (finalBossController == null)
        {
            finalBossController = FindFirstObjectByType<PracticeFinalBossController>();
        }
    }

    private void SubscribeEvents()
    {
        if (gameStateController != null)
        {
            gameStateController.StateChanged -= HandleGameStateChanged;
            gameStateController.StateChanged += HandleGameStateChanged;
        }

        if (runDirector != null)
        {
            runDirector.RunPhaseChanged -= HandleRunPhaseChanged;
            runDirector.RunPhaseChanged += HandleRunPhaseChanged;
        }

        if (finalBossController != null)
        {
            finalBossController.BossSpawned -= HandleBossSpawned;
            finalBossController.BossSpawned += HandleBossSpawned;
        }
    }

    private void UnsubscribeEvents()
    {
        if (gameStateController != null)
        {
            gameStateController.StateChanged -= HandleGameStateChanged;
        }

        if (runDirector != null)
        {
            runDirector.RunPhaseChanged -= HandleRunPhaseChanged;
        }

        if (finalBossController != null)

using UnityEngine;

public class PracticePlayerAudioFeedback : HaiMonoBehaviour
{
    [SerializeField]
    protected PracticePlayerHealth playerHealth;

    protected override void LoadComponents()
    {
        base.LoadComponents();

        LoadPlayerHealth();
    }

    private void LoadPlayerHealth()
    {
        if (playerHealth != null)
        {
            return;
        }

        playerHealth =
            GetComponent<PracticePlayerHealth>();
    }

    protected override void OnEnable()
    {
        base.OnEnable();

        LoadPlayerHealth();
        SubscribePlayerEvents();
    }

    protected override void OnDisable()
    {
        UnsubscribePlayerEvents();

        base.OnDisable();
    }

    private void SubscribePlayerEvents()
    {
        if (playerHealth == null)
        {
            return;
        }

        playerHealth.DamageTaken +=
            HandlePlayerDamageTaken;
    }

    private void UnsubscribePlayerEvents()
    {
        if (playerHealth == null)
        {
            return;
        }

        playerHealth.DamageTaken -=
            HandlePlayerDamageTaken;
    }

    private void HandlePlayerDamageTaken(float damageAmount)
    {
        PracticeAudioManager.Instance?.
            PlayPlayerHitSfx();

using UnityEngine;

public class PracticeCameraShake2D : HaiMonoBehaviour
{
    public static PracticeCameraShake2D Instance { get; private set; }

    [Header("References")]
    [SerializeField] private PracticeFinalBossController finalBossController;
    [SerializeField] private PracticeGameStateController gameStateController;
    [SerializeField] private PracticePlayerLevelSystem playerLevelSystem;

    [Header("Shake Settings")]
    [SerializeField, Min(0f)] private float maxShakeOffset = 0.45f;
    [SerializeField, Min(1f)] private float shakeFrequency = 28f;

    private Vector3 initialLocalPosition;
    private bool hasCachedInitialPosition;
    private float remainingShakeTime;
    private float currentShakeDuration;
    private float currentShakeMagnitude;
    private float noiseSeed;

    protected override void LoadComponents()
    {
        base.LoadComponents();
        LoadSceneSystems();
    }

    protected override void Awake()
    {
        if (Instance != null && Instance != this)
        {
            Destroy(this);
            return;
        }

        Instance = this;
        base.Awake();

        CacheInitialPosition();
        noiseSeed = Random.Range(0f, 100f);
        LoadSceneSystems();
    }

    protected override void OnEnable()
    {
        base.OnEnable();
        CacheInitialPosition();
        LoadSceneSystems();
        SubscribeEvents();
    }

    protected override void OnDisable()
    {
        UnsubscribeEvents();
        ResetCameraPosition();
        base.OnDisable();
    }

    private void OnDestroy()
    {
        if (Instance == this)
        {
            Instance = null;
        }
    }

    private void CacheInitialPosition()
    {
        if (hasCachedInitialPosition)
        {
            return;
        }

        initialLocalPosition = transform.localPosition;
        hasCachedInitialPosition = true;
    }

    private void LoadSceneSystems()
    {
        if (finalBossController == null)
        {
            finalBossController = FindFirstObjectByType<PracticeFinalBossController>();
        }

        if (gameStateController == null)
        {
            gameStateController = FindFirstObjectByType<PracticeGameStateController>();
        }

        if (playerLevelSystem == null)
        {
            playerLevelSystem = FindFirstObjectByType<PracticePlayerLevelSystem>();
        }
    }

    private void SubscribeEvents()
    {
        if (finalBossController != null)
        {
            finalBossController.BossSpawned -= HandleBossSpawned;
            finalBossController.BossSpawned += HandleBossSpawned;

            finalBossController.BossDefeated -= HandleBossDefeated;
            finalBossController.BossDefeated += HandleBossDefeated;
        }

        if (gameStateController != null)
        {
            gameStateController.StateChanged -= HandleGameStateChanged;
            gameStateController.StateChanged += HandleGameStateChanged;
        }

        if (playerLevelSystem != null)
        {
            playerLevelSystem.LevelChanged -= HandlePlayerLevelUp;
            playerLevelSystem.LevelChanged += HandlePlayerLevelUp;
        }
